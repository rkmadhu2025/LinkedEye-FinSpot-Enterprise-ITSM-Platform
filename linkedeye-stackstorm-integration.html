LinkedEye-FinSpot Actions

actions/linkedeye.create_incident.yaml
name: create_incident pack: linkedeye description: Create an incident in LinkedEye-FinSpot runner_type: python-script enabled: true entry_point: create_incident.py parameters: short_description: type: string description: Brief summary of the incident required: true description: type: string description: Detailed description required: false priority: type: string enum: ['Low', 'Medium', 'High', 'Critical'] default: 'Medium' category: type: string enum: ['Hardware', 'Software', 'Network', 'Infrastructure'] assignment_group: type: string description: Team to assign incident to
actions/create_incident.py
import requests from st2common.runners.base_action import Action class CreateIncidentAction(Action): def run(self, short_description, description="", priority="Medium", category="", assignment_group=""): api_url = self.config['api_url'] api_token = self.config['api_token'] headers = { 'Authorization': f'Bearer {api_token}', 'Content-Type': 'application/json' } payload = { 'short_description': short_description, 'description': description, 'priority': priority, 'category': category, 'assignment_group': assignment_group } response = requests.post( f'{api_url}/incidents', headers=headers, json=payload ) if response.status_code == 201: return (True, response.json()) else: return (False, response.text)
